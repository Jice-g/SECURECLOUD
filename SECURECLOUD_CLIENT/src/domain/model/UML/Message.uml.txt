namespace Domain {

/// @brief Entity representing a single instant message.
///
/// A Message is a child entity of the Conversation aggregate. It cannot
/// exist outside a conversation. It is created in BUFFERED state and
/// follows a strict lifecycle.
class Message {

    // ---- Attributes ----
    - id             : MessageId
    - conversationId : ConversationId
    - senderId       : UserId
    - content        : MessageContent
    - createdAt      : Timestamp
    - priority       : Priority
    - status         : MessageStatus
    - attachments    : List<AttachmentId>   // References, not full objects

    // ---- Constructor ----
    /// @brief Constructs a new Message in BUFFERED state.
    /// @param id, conversationId, senderId, content, priority
    /// @throws std::invalid_argument if id, conversationId or senderId is empty
    + Message(id : MessageId,
              conversationId : ConversationId,
              senderId : UserId,
              content : MessageContent,
              priority : Priority)

    // ---- Getters (read-only) ----
    + id()             : MessageId
    + conversationId() : ConversationId
    + senderId()       : UserId
    + content()        : MessageContent
    + createdAt()      : Timestamp
    + priority()       : Priority
    + status()         : MessageStatus
    + attachments()    : List<AttachmentId>

    // ---- State machine (business behavior) ----
    /// Transitions BUFFERED -> SENDING
    + markAsSending() : void
    /// Transitions SENDING -> SENT
    + markAsSent() : void
    /// Transitions SENT -> DELIVERED
    + markAsDelivered() : void
    /// Transitions SENDING or SENT -> UNDELIVERED
    + markAsUndelivered() : void

    // ---- Content editing ----
    /// Only allowed if status == BUFFERED
    + editContent(newContent : MessageContent) : void

    // ---- Attachment management ----
    + addAttachment(id : AttachmentId) : void
    + removeAttachment(id : AttachmentId) : void
    + hasAttachments() : Boolean

    // ---- Validation ----
    /// @return true if the message satisfies its own invariants.
    /// @note Called before persistence by the Application layer.
    + isValid() : Boolean

} // class Message

} // namespace Domain/Model


// ========================= CONSTRAINTS =========================
//
// [C1] INVARIANT - Content OR Attachment
//      A valid Message must have non-empty content OR at least one
//      attachment. Both empty is forbidden.
//
// [C2] INVARIANT - State machine is strict
//      BUFFERED -> SENDING -> SENT -> DELIVERED
//                       \______/
//                           \--> UNDELIVERED
//      A transition not listed above must throw std::logic_error.
//
// [C3] INVARIANT - Initial state is BUFFERED
//      A newly constructed Message is always in BUFFERED state.
//
// [C4] PRECONDITION - editContent()
//      Can only be called when status == BUFFERED. Editing a
//      message that has been sent is a business rule violation.
//
// [C5] INVARIANT - createdAt is immutable
//      The timestamp is set once at construction and never changes.
//
// [C6] INVARIANT - No duplicate attachment
//      The same AttachmentId cannot be added twice.
//
// ==============================================================



//
// Value Object :
// ----------------------

namespace Domain/Model {

class Attachment {
    - id        : AttachmentId
    - fileName  : String
    - sizeBytes : Integer
    - mimeType  : String
    + Attachment(id : AttachmentId, fileName : String, sizeBytes : Integer, mimeType : String)
    + id()        : AttachmentId
    + fileName()  : String
    + sizeBytes() : Integer
    + mimeType()  : String
    + isImage()    : Boolean
    + isDocument() : Boolean
}

/*

Rules the constructor enforces:

    fileName is non-empty.

    sizeBytes > 0.

    mimeType contains a / (basic type/subtype check).
*/


namespace Domain/Model {


class MessageContent {
    - text : String                       // trimmed at construction
    + MessageContent(text : String)       // trims, allows empty, max 4096
    + value()  : String
    + isEmpty() : Boolean
    + length() : Integer
}

/*
Key decision: Empty content is allowed at the MessageContent level. The Message entity enforces "content OR attachment". This separation is intentional — MessageContent doesn't know about attachments.
*/
