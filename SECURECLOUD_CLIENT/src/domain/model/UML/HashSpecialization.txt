// =============================================================================
// std::hash specializations
// =============================================================================
//
// These specializations allow the ID types to be used as keys in
// std::unordered_map and std::unordered_set. They must live in the
// `std` namespace (a rare, legal exception for user-defined types).

namespace std {

/// @brief Hash specialization for domain::UserId.
template <>
struct hash<domain::UserId> {
    std::size_t operator()(const domain::UserId& id) const noexcept {
        return std::hash<std::string>{}(id.value());
    }
};

} // namespace std
